# minecraft-pack-1201 - ИНСТРУКЦИЯ ЗАГРУЗКИ (формат 1)
#
# Это стабильный указатель для игроков. В их файлах адресов пака нет: скрипт
# читает этот файл по двум неизменным адресам и берёт первый доступный источник:
#   http://26.26.205.227:8787/bootstrap.txt    (ПК владельца, Radmin - быстрее)
#   https://raw.githubusercontent.com/CrystalysGolem/minecraft-pack-1201/main/bootstrap.txt
#
# Формат: строки pack=<адрес pack.toml>, порядок = приоритет (сверху вниз).
# Строки, начинающиеся с #, игнорируются. Клиент берёт первый РАБОЧИЙ источник,
# а успешной загрузкой инструкции считает файл, где есть хотя бы одна строка pack=
# (поэтому страница ошибки прокси за инструкцию не сойдёт).
#
# Меняешь хостинг или порядок источников - правишь только этот файл, игрокам
# обновлять свои файлы НЕ нужно. Свои строки pack= сохраняются при перегенерации.
# Примерный порядок для правки: быстрый локальный источник, затем публичный.
format=1
pack=http://26.26.205.227:8787/pack.toml
pack=https://raw.githubusercontent.com/CrystalysGolem/minecraft-pack-1201/main/pack.toml
